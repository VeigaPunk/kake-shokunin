"""Canned do-nothing control (CriticClaims r84w1): prints the frozen CLI lines
for the three fixture pairs this round freezes. NOT a lane instrument; lives

import argparse
import sys
from pathlib import Path

CONTROL = "capture-control/control/a"

KEY = {
    REAL: ("", ("surface=ds4cc-primary changed=0 reordered=0 numeric_parameter=0 directive_added=0 directive_removed=0 directive_changed=0 header_added=0 header_removed=0 unclassified=0 verdict=policy-unchanged", "surface=ds4cc-robots changed=0 reordered=0 numeric_parameter=0 directive_added=0 directive_removed=0 directive_changed=0 header_added=0 header_removed=0 unclassified=0 verdict=policy-unchanged", "surface=ds4cc-sitemap changed=0 reordered=0 numeric_parameter=0 directive_added=0 directive_removed=0 directive_changed=0 header_added=0 header_removed=0 unclassified=0 verdict=policy-unchanged"), "summary surfaces=3 changed=0 reordered=0 numeric_parameter=0 directive_added=0 directive_removed=0 directive_changed=0 header_added=0 header_removed=0 unclassified=0 verdict=policy-unchanged"),
    CONTROL: ("change surface=ds4cc-primary header=cache-control class=numeric_parameter\nchange surface=ds4cc-robots header=x-frame-options class=header_added\nchange surface=ds4cc-sitemap header=referrer-policy class=reordered\n", ("surface=ds4cc-primary changed=1 reordered=0 numeric_parameter=1 directive_added=0 directive_removed=0 directive_changed=0 header_added=0 header_removed=0 unclassified=0 verdict=policy-changed", "surface=ds4cc-robots changed=1 reordered=0 numeric_parameter=0 directive_added=0 directive_removed=0 directive_changed=0 header_added=1 header_removed=0 unclassified=0 verdict=policy-changed", "surface=ds4cc-sitemap changed=1 reordered=1 numeric_parameter=0 directive_added=0 directive_removed=0 directive_changed=0 header_added=0 header_removed=0 unclassified=0 verdict=policy-rewritten-equivalent"), "summary surfaces=3 changed=3 reordered=1 numeric_parameter=1 directive_added=0 directive_removed=0 directive_changed=0 header_added=1 header_removed=0 unclassified=0 verdict=policy-changed"),
    EQUIV: ("change surface=ds4cc-sitemap header=referrer-policy class=reordered\n", ("surface=ds4cc-primary changed=0 reordered=0 numeric_parameter=0 directive_added=0 directive_removed=0 directive_changed=0 header_added=0 header_removed=0 unclassified=0 verdict=policy-unchanged", "surface=ds4cc-robots changed=0 reordered=0 numeric_parameter=0 directive_added=0 directive_removed=0 directive_changed=0 header_added=0 header_removed=0 unclassified=0 verdict=policy-unchanged", "surface=ds4cc-sitemap changed=1 reordered=1 numeric_parameter=0 directive_added=0 directive_removed=0 directive_changed=0 header_added=0 header_removed=0 unclassified=0 verdict=policy-rewritten-equivalent"), "summary surfaces=3 changed=1 reordered=1 numeric_parameter=0 directive_added=0 directive_removed=0 directive_changed=0 header_added=0 header_removed=0 unclassified=0 verdict=policy-rewritten-equivalent"),
}


def render(which: str) -> str:
    changes, surfaces, summary = KEY[which]
    return changes + "\n".join(surfaces) + "\n" + summary + "\n"


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description=__doc__.splitlines()[0])
    parser.add_argument("--a", type=Path)
    parser.add_argument("--b", type=Path)
    parser.add_argument("--declared", type=Path)
    parser.add_argument("--selftest", action="store_true")
    parser.add_argument("--work", type=Path)
    args = parser.parse_args(argv)
    if args.selftest:
        print("ok   emits-frozen-cli-lines")
        print("ok   offline")
        print("selftest: 2/2 passed")
        return 0
    if not args.a or not args.b or not args.declared:
        parser.print_usage(sys.stderr)
        return 64
    a = str(args.a)
    for probe in (REAL, CONTROL, EQUIV):
        if probe in a:
            sys.stdout.write(render(probe))
            return 0
    print("fault: unevaluable input", file=sys.stderr)
    return 1


if __name__ == "__main__":
    raise SystemExit(main())
