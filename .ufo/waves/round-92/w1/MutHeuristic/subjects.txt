#!/usr/bin/env python3
"""Heuristic freshness decision instrument — contract v1 (round 92, kake).

Decides whether a heuristic freshness lifetime may be assigned to a stored
response, and what that lifetime is. RFC 9111 §4.2.2, over supplied integer
epoch seconds only: no clock, no HTTP-date parser, no network, no cache store.

Lane conventions (NOT RFC requirements): the 10 % fraction and the 86400-second
cap. The RFC encourages "no more than some fraction of the interval" and calls
10 % a typical setting; it names no cap.

Output: one JSON object with exactly `decision`, `lifetime`, `heuristic`; exit 0.
Refusal: `invalid_input: <reason>` on stderr, exit 1, nothing on stdout.
"""

import json
import re
import sys

HEURISTICALLY_CACHEABLE = frozenset(
    (200, 203, 204, 206, 300, 301, 308, 404, 405, 410, 414, 501))
DELTA = ("max-age", "s-maxage")
INT_RE = re.compile(r"[0-9]+")


class Invalid(Exception):
    pass


def ascii_lower(s):
    return "".join(chr(ord(c) + 32) if "A" <= c <= "Z" else c for c in s)


def is_int(v):
    return isinstance(v, int) and not isinstance(v, bool)


def parse_cc(value):
    out = {}
    for raw in value.split(","):
        d = raw.strip(" \t")
        if not d:
            raise Invalid("empty Cache-Control directive")
        name, sep, arg = d.partition("=")
        name = ascii_lower(name.strip(" \t"))
        if not name:
            raise Invalid("empty Cache-Control directive name")
        if sep:
            arg = arg.strip(" \t")
            if name in DELTA:
                if not INT_RE.fullmatch(arg):
                    raise Invalid("delta-seconds argument is not a bare "
                                  "non-negative integer")
                arg = int(arg)
            elif len(arg) >= 2 and arg[0] == '"' and arg[-1] == '"':
                arg = arg[1:-1]
        else:
            arg = None
        if name in out and out[name] != arg:
            raise Invalid("conflicting Cache-Control directive %s" % name)
        out[name] = arg
    return out


def norm_headers(raw):
    if not isinstance(raw, dict):
        raise Invalid("headers is not an object")
    seen = set()
    out = {}
    for name, value in raw.items():
        if not isinstance(name, str) or not isinstance(value, str):
            raise Invalid("header name or value is not a string")
        if "\r" in value or "\n" in value or "\0" in value:
            raise Invalid("header value contains CR, LF or NUL")
        low = ascii_lower(name)
        if low in seen:
            raise Invalid("duplicate header name")
        seen.add(low)
        out[low] = value.strip(" \t")
    return out


def int_header(headers, name, required=False):
    v = headers.get(name)
    if v is None:
        if required:
            raise Invalid("%s is absent" % name)
        return None
    if not INT_RE.fullmatch(v):
        raise Invalid("%s is not a decimal integer" % name)
    return int(v)


def decide(inp):
    if not isinstance(inp, dict):
        raise Invalid("input is not an object")
    if set(inp) - {"stored", "now", "shared"}:
        raise Invalid("unexpected top-level key")
    stored = inp.get("stored")
    if not isinstance(stored, dict):
        raise Invalid("stored is not an object")
    if set(stored) - {"status", "headers"}:
        raise Invalid("unexpected stored key")
    status = stored.get("status")
    if not is_int(status) or not 100 <= status <= 599:
        raise Invalid("status is not an int in 100..599")
    now = inp.get("now")
    if not is_int(now) or now < 0:
        raise Invalid("now is not a non-negative int")
    shared = inp.get("shared", False)
    if not isinstance(shared, bool):
        raise Invalid("shared is not a bool")

    headers = norm_headers(stored.get("headers", {}))
    cc = parse_cc(headers["cache-control"]) if "cache-control" in headers else {}
    date = int_header(headers, "date", required=True)
    expires = int_header(headers, "expires")
    lastmod = int_header(headers, "last-modified")

    # clause 9 — not-storable
    if "no-store" in cc:
        return {"decision": "not-storable", "lifetime": None, "heuristic": False}
    if shared and "private" in cc:
        return {"decision": "not-storable", "lifetime": None, "heuristic": False}
    if "vary" in headers and any(
            m.strip(" \t") == "*" for m in headers["vary"].split(",")):
        return {"decision": "not-storable", "lifetime": None, "heuristic": False}

    # clause 10 — explicit expiration wins; heuristics are never used
    if shared and cc.get("s-maxage") is not None:
        return {"decision": "explicit", "lifetime": cc["s-maxage"], "heuristic": False}
    if cc.get("max-age") is not None:
        return {"decision": "explicit", "lifetime": cc["max-age"], "heuristic": False}
    if expires is not None:
        return {"decision": "explicit", "lifetime": expires - date, "heuristic": False}

    # clause 11 — the cacheability gate
    if status not in HEURISTICALLY_CACHEABLE and "public" not in cc:
        return {"decision": "not-heuristically-cacheable", "lifetime": None,
                "heuristic": False}

    # clause 12 / 13 — lane conventions: 10 % fraction, 86400 s cap
    if lastmod is not None and lastmod <= date:
        return {"decision": "heuristic",
                "lifetime": min((date - lastmod) // 10, 86400), "heuristic": True}
    return {"decision": "heuristic-undecided", "lifetime": None, "heuristic": False}


def main(argv):
    if argv and argv[0] == "--selftest":
        return selftest()
    try:
        raw = sys.stdin.read()
        inp = json.loads(raw)
    except ValueError:
        sys.stderr.write("invalid_input: input is not JSON\n")
        return 1
    try:
        out = decide(inp)
    except Invalid as exc:
        sys.stderr.write("invalid_input: %s\n" % exc)
        return 1
    sys.stdout.write(json.dumps(out, sort_keys=True) + "\n")
    return 0


CASES = [
    ({"stored": {"status": 200, "headers": {"Date": "1000000",
                                            "Last-Modified": "990000"}},
      "now": 1000000}, {"decision": "heuristic", "lifetime": 1000, "heuristic": True}),
    ({"stored": {"status": 200, "headers": {"Date": "1000000",
                                            "Last-Modified": "100000"}},
      "now": 1000000}, {"decision": "heuristic", "lifetime": 86400, "heuristic": True}),
    ({"stored": {"status": 200, "headers": {"Date": "1000000",
                                            "Cache-Control": "max-age=600"}},
      "now": 1000000}, {"decision": "explicit", "lifetime": 600, "heuristic": False}),
    ({"stored": {"status": 500, "headers": {"Date": "1000000"}},
      "now": 1000000}, {"decision": "not-heuristically-cacheable", "lifetime": None,
                        "heuristic": False}),
    ({"stored": {"status": 200, "headers": {"Date": "1000000",
                                            "Cache-Control": "no-store"}},
      "now": 1000000}, {"decision": "not-storable", "lifetime": None, "heuristic": False}),
]


def selftest():
    ok = 0
    for inp, want in CASES:
        got = decide(inp)
        if got != want:
            sys.stderr.write("selftest FAIL %r != %r\n" % (got, want))
        else:
            ok += 1
    print("selftest %d/%d" % (ok, len(CASES)))
    return 0 if ok == len(CASES) else 1


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
