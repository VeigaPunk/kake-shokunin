def git(*a, check=True):
    p = subprocess.run(["git", "-C", FX, *a], capture_output=True, text=True)
    if check and p.returncode:
        raise SystemExit(f"git {a} rc={p.returncode} {p.stderr}")
    return p

def rev_of(x):
    return git("rev-parse", "--verify", x + "^{commit}").stdout.strip()

def _window(ls):
    a = ls.index("## Capabilities")
    b = next(i for i in range(a, len(ls)) if ls[i].startswith("### Capability rows without"))
    return a, b

def one(s, prefix, old, new):
    lines = s.split("\n")
    a, b = _window(lines)
    hits = [i for i in range(a, b) if lines[i].startswith(prefix)]
    assert len(hits) == 1, (prefix, len(hits))
    assert lines[hits[0]].count(old) >= 1, (prefix, old)
    lines[hits[0]] = lines[hits[0]].replace(old, new, 1)
    return "\n".join(lines)

def m_base(s): return s

def m_escpipe(s):  # same as prefix, name written with an escaped pipe

def m_unrelated_ev(s):  # existing unrelated evidence path + entry path kept in prose

def _lines(s): return s.split("\n")
def _ins_before(s, anchor_prefix, new_lines):
    ls = _lines(s); h = [i for i, l in enumerate(ls) if l.startswith(anchor_prefix)]
    assert len(h) == 1, anchor_prefix
    return "\n".join(ls[:h[0]] + new_lines + ls[h[0]:])
def m_phantom_sub(s):  # phantom placed in the ### subsection the repair created
    return _ins_before(s, "### Capability rows without", ["", "| Capability | What it can now do | Evidence pointer | First-working round |", "|---|---|---|---|", PH, ""])
def m_phantom_first(s):  # phantom in the first table
    ls = _lines(s); h = ls.index("| Capability | What it can now do | Evidence pointer | First-working round |")
    return "\n".join(ls[:h + 2] + [PH] + ls[h + 2:])
def m_c2_row(s):  # the C2 row, round 98, as the frozen predicate requires
    ls = _lines(s); h = ls.index("| Capability | What it can now do | Evidence pointer | First-working round |")
    row = "| Stale-serving extensions (RFC 5861 stale-while-revalidate, stale-if-error) | Decides, over supplied Cache-Control and freshness values only, whether a stored response may be served stale for revalidation or on error, and refuses malformed input | `.ufo/waves/round-98/c98_stale.py` | 98 |"
    return "\n".join(ls[:h + 2] + [row] + ls[h + 2:])
def m_fenced_only(s):
    ls = _lines(s); h = ls.index("| Capability | What it can now do | Evidence pointer | First-working round |")
    e = h
    while e < len(ls) and ls[e].startswith("|"): e += 1
    return "\n".join(ls[:h] + ["```"] + ls[h:e] + ["```"] + ls[e:])
def m_fenced_decoy(s):
    ls = _lines(s); h = ls.index("| Capability | What it can now do | Evidence pointer | First-working round |")
    return "\n".join(ls[:h] + ["```", "| Capability | What it can now do | Evidence pointer | First-working round |", "|---|---|---|---|", "| Decoy | Decoy row claiming nothing about any table at all here | `SCORECARD.md` | 1 |", "```"] + ls[h:])
def m_inplace(s):  # existing matched row's evidence changed to an existing unrelated path
    return one(s, "| Revalidation exchange, contract v1 |", "`scripts/reval_exchange.py`", "`SCORECARD.md`; scripts/reval_exchange.py")

                            "--fleet-revision", FLEET_REV, "--fleet", "/home/vgpnk/Projects/shokunin"],
