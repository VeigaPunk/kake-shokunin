def gen_outs(m, seed, n=240):

    return cases

def seed_cases(seed): return ORA_GEN(seed, 240)
def lottery(m, seeds):
    passing = 0; mism_cases = 0; total = 0
    for sd in seeds:
        cases = seed_cases(sd); bad = 0
        for c in cases:
            total += 1
            if outcome(m, c) != outcome(ORA, c): bad += 1
        mism_cases += bad; passing += bad == 0
    return passing, len(seeds), mism_cases, total
